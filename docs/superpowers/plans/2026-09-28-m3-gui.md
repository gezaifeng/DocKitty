# M3 GUI 主体 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 在 M1/M2 的 packages/core 引擎之上构建 Electron + React 桌面应用，实现仓库管理、引导式编辑闭环、时间线、文档预览，让非程序员可独立完成全流程。

**Architecture:** Electron 主进程直接调用 packages/core（零网络层），通过 contextBridge 最小化白名单暴露 IPC API；渲染进程为 React 19 SPA，使用 TanStack Query 管理异步数据、Zustand 管理全局客户端状态；packages/shared 持有 zod 校验的 IPC 契约，主进程/预加载/渲染进程三方共享类型；chokidar v4 在主进程监听工作区文件，通过 ipcMain → webContents.send 推送保存事件。

**Tech Stack:** Electron 32+ · electron-vite 3+ (Vite 7) · React 19 · TypeScript 5.7 · shadcn/ui · Tailwind CSS v4 · Zustand 5 · TanStack Query v5 · chokidar v4 · docx-preview · ExcelJS · zod 3 · Vitest 3 · @testing-library/react 16

**Spec:** `docs/superpowers/specs/2026-09-28-word-git-design.md`

## Global Constraints

- pnpm workspaces monorepo；新增 `apps/desktop` + `packages/shared`，不修改 packages/core 或 packages/cli
- TypeScript strict ESM；`packages/shared` 源文件使用 `.js` import 扩展名（NodeNext）；`apps/desktop` 主进程/预加载使用 `.js` 扩展名；渲染进程使用 Bundler moduleResolution（无需 `.js` 扩展名）
- 所有 IPC handler 的业务逻辑函数单独 export 供测试直接调用，不依赖 ipcMain
- contextIsolation: true，sandbox: true；渲染进程只通过 `window.wordgit` 访问 IPC
- 中文 UI 文字，英文代码标识符
- 不引入加密；不实现网络同步；不实现自研文档编辑器
- 开发分支：`m3-gui`（从 `m2-reliability` 创建）
- electron-vite dev 模式无需手动构建步骤：`pnpm dev` 直接热更新
- 主进程测试用 vitest + Node 环境；渲染进程测试用 vitest + jsdom + @testing-library/react
- 每个 task 完成后 `tsc --noEmit` 必须通过

// packages/shared/src/schemas.ts
import { z } from 'zod'

export const RepoCreateReq = z.object({ root: z.string().min(1), name: z.string().min(1) })
export const RepoCreateRes = z.object({ repoRoot: z.string() })

export const RepoOpenReq = z.object({ root: z.string().min(1) })
export const RepoOpenRes = z.object({ repoRoot: z.string(), repoName: z.string() })

export const DocImportReq = z.object({ sourcePath: z.string().min(1), name: z.string().optional() })
export const DocImportRes = z.object({ docId: z.string(), commitId: z.string() })

export const DocInfoSchema = z.object({
  docId: z.string(),
  docName: z.string(),
  currentBranch: z.string(),
  branches: z.array(z.string()),
  commitCount: z.number().int().nonnegative(),
  lastCommitAt: z.string().nullable(),
})

export const DocEditReq = z.object({ docName: z.string().min(1) })
export const DocEditRes = z.object({ workspacePath: z.string() })

export const CommitCreateReq = z.object({
  docName: z.string().min(1),
  message: z.string().min(1),
  author: z.string().default('user'),
  branch: z.string().optional(),
})

export const CommitResultSchema = z.object({
  docId: z.string(),
  branch: z.string(),
  commitId: z.string().nullable(),
  changed: z.array(z.string()),
})

export const CommitNodeSchema = z.object({
  id: z.string(),
  parents: z.array(z.string()),
  message: z.string(),
  author: z.string(),
  createdAt: z.string(),
})

export const CommitListReq = z.object({ docName: z.string().min(1), branch: z.string().optional() })
export const CommitCheckoutReq = z.object({
  docName: z.string().min(1),
  commitId: z.string().optional(),
  branch: z.string().optional(),
})
export const CommitCheckoutRes = z.object({ docId: z.string(), commitId: z.string() })

export const BranchCreateReq = z.object({
  docName: z.string().min(1),
  branch: z.string().min(1),
  fromBranch: z.string().optional(),
  fromCommitId: z.string().optional(),
})
export const BranchSwitchReq = z.object({ docName: z.string().min(1), branch: z.string().min(1) })
export const BranchDeleteReq = z.object({ docName: z.string().min(1), branch: z.string().min(1) })

export const QuarantineEntrySchema = z.object({
  id: z.string(),
  docId: z.string(),
  docName: z.string(),
  branch: z.string(),
  commitId: z.string(),
  deletedAt: z.string(),
})

export const StatsStorageRes = z.object({
  totalBytes: z.number().nonnegative(),
  objectCount: z.number().int().nonnegative(),
  docCount: z.number().int().nonnegative(),
  versionCount: z.number().int().nonnegative(),
})

export const BackupExportReq = z.object({ destPath: z.string().min(1) })
export const BackupExportRes = z.object({ fileCount: z.number().int() })
export const BackupImportReq = z.object({ srcPath: z.string().min(1), destRoot: z.string().min(1) })

// apps/desktop/src/main/ipc/docHandlers.ts
import path from 'node:path'
import { ipcMain, shell } from 'electron'
import {
  importDoc as coreImportDoc,
  listCommits,
  checkoutDoc,
  docIds,
  listBranches,
  readCurrentBranch,
  docIdOf,
} from '@wordgit/core'
import {
  CH,
  DocImportReq, DocInfoSchema, DocEditReq,
} from '@wordgit/shared'
import type { DocInfo } from '@wordgit/shared'
import { getRepo } from '../repoState.js'
import { watchDoc } from '../watcher.js'

export async function listDocs(_args: unknown): Promise<DocInfo[]> {
  const repo = getRepo()
  const names = await docIds(repo)
  const out: DocInfo[] = []
  for (const docName of names) {
    const id = docIdOf(docName)
    const [branches, currentBranch, commits] = await Promise.all([
      listBranches(repo, id),
      readCurrentBranch(repo, id),
      listCommits(repo, docName),
    ])
    out.push(DocInfoSchema.parse({
      docId: id,
      docName,
      currentBranch,
      branches,
      commitCount: commits.length,
      lastCommitAt: commits[0]?.createdAt ?? null,
    }))
  }
  return out
}

export async function importDocHandler(args: unknown): Promise<{ docId: string; commitId: string }> {
  const { sourcePath, name } = DocImportReq.parse(args)
  const repo = getRepo()
  return coreImportDoc(repo, sourcePath, name)
}

export async function editDoc(args: unknown): Promise<{ workspacePath: string }> {
  const { docName } = DocEditReq.parse(args)
  const repo = getRepo()
  await checkoutDoc(repo, docName, {})
  const workspacePath = path.join(repo.root, 'workspace', docName)
  await shell.openPath(workspacePath)
  watchDoc(docName, workspacePath)
  return { workspacePath }
}

export function registerDocHandlers(): void {
  ipcMain.handle(CH.DOC_LIST, (_e, arg: unknown) => listDocs(arg))
  ipcMain.handle(CH.DOC_IMPORT, (_e, arg: unknown) => importDocHandler(arg))

import { promises as fs } from 'node:fs'
import { ipcMain } from 'electron'
import {
  scrub as coreScrub,
  listQuarantine,
  restoreFromQuarantine,
  purgeQuarantine,
  exportBackup,
  importBackup,
  docIds,
  listCommits,
} from '@wordgit/core'
import {
  CH,
  QuarantineRestoreReq, QuarantinePurgeReq,
  BackupExportReq, BackupImportReq,
  StatsStorageRes,
} from '@wordgit/shared'
import type { StorageStats, ScrubResult, QuarantineEntry } from '@wordgit/shared'
import { getRepo, setRepo } from '../repoState.js'

async function sumDirBytes(dir: string): Promise<{ bytes: number; count: number }> {
  let bytes = 0
  let count = 0
  try {
    const entries = await fs.readdir(dir, { recursive: true, withFileTypes: true })
    for (const e of entries) {
      if (!e.isFile()) continue
      const parentPath = (e as NodeJS.Dirent & { parentPath?: string }).parentPath ?? dir
      try {
        const st = await fs.stat(path.join(parentPath, e.name))
        bytes += st.size
        count++
      } catch { /* skip */ }
    }
  } catch { /* dir missing */ }
  return { bytes, count }
}

export async function getStorageStats(_args: unknown): Promise<StorageStats> {
  const repo = getRepo()
  const [blobs, chunks, ooxml] = await Promise.all([
    sumDirBytes(repo.objectsDir('blobs')),
    sumDirBytes(repo.objectsDir('chunks')),
    sumDirBytes(repo.objectsDir('ooxml')),
  ])
  const names = await docIds(repo)
  let versionCount = 0
  for (const name of names) {
    const commits = await listCommits(repo, name)
    versionCount += commits.length
  }
  return StatsStorageRes.parse({

// apps/desktop/src/main/ipc/previewHandlers.ts
import path from 'node:path'
import { promises as fs } from 'node:fs'
import { ipcMain } from 'electron'
import {
  StorageContext, readCommit, readTree, getManifest, materializeFile,
} from '@wordgit/core'
import { CH, PreviewGetBytesReq } from '@wordgit/shared'
import { getRepo } from '../repoState.js'

export async function getPreviewBytes(
  args: unknown,
): Promise<{ data: ArrayBuffer; ext: string }> {
  const { docName, commitId } = PreviewGetBytesReq.parse(args)
  const repo = getRepo()
  const ext = path.extname(docName).toLowerCase().replace('.', '')

  let raw: Uint8Array
  if (commitId) {
    const ctx = StorageContext.forRepo(repo)
    const commit = await readCommit(ctx, commitId)
    const tree = await readTree(ctx, commit.tree)
    // single-file doc: entry key is '' (empty string)
    const manifestId = tree.entries[''] ?? Object.values(tree.entries)[0]
    if (!manifestId) throw new Error(`提交 ${commitId} 中没有可预览的文件`)
    const manifest = await getManifest(ctx, manifestId)
    raw = await materializeFile(ctx, manifest)
  } else {
    const wsPath = path.join(repo.root, 'workspace', docName)
    raw = new Uint8Array(await fs.readFile(wsPath))
  }

  // Transfer as ArrayBuffer
  const data = raw.buffer.slice(raw.byteOffset, raw.byteOffset + raw.byteLength) as ArrayBuffer
  return { data, ext }
}

export function registerPreviewHandlers(): void {
  ipcMain.handle(CH.PREVIEW_GET_BYTES, (_e, arg: unknown) => getPreviewBytes(arg))
}

// apps/desktop/src/main/watcher.ts
import { watch, type FSWatcher } from 'chokidar'
import type { BrowserWindow } from 'electron'
import { CH } from '@wordgit/shared'
import type { WatcherChangeEvent } from '@wordgit/shared'

let _watcher: FSWatcher | null = null
let _mainWindow: BrowserWindow | null = null

export function setMainWindow(win: BrowserWindow): void {
  _mainWindow = win
}

export function watchDoc(docName: string, workspacePath: string): void {
  if (_watcher) {
    _watcher.close()
    _watcher = null
  }
  _watcher = watch(workspacePath, {
    ignoreInitial: true,
    depth: 0,
    awaitWriteFinish: { stabilityThreshold: 500, pollInterval: 100 },
  })
  _watcher.on('change', () => {
    if (!_mainWindow) return
    const event: WatcherChangeEvent = { docName, workspacePath }
    _mainWindow.webContents.send(CH.WATCHER_CHANGE, event)
  })
}

export function stopWatcher(): void {
  _watcher?.close()
  _watcher = null
}

// apps/desktop/src/renderer/src/pages/FirstRunPage.tsx
import { useState } from 'react'
import { api } from '@renderer/lib/ipc'
import { useRepoStore } from '@renderer/store/repoStore'
import { Button } from '@renderer/components/ui/button'
import {
  Card,
  CardContent,
  CardHeader,
  CardTitle,
} from '@renderer/components/ui/card'
import { Input } from '@renderer/components/ui/input'
import { Label } from '@renderer/components/ui/label'
import path from 'node:path'

type Step = 'landing' | 'naming'

export function FirstRunPage() {
  const { setRepo } = useRepoStore()
  const [step, setStep] = useState<Step>('landing')
  const [chosenDir, setChosenDir] = useState<string>('')
  const [repoName, setRepoName] = useState<string>('')
  const [error, setError] = useState<string | null>(null)
  const [loading, setLoading] = useState(false)

  async function handleCreate() {
    setError(null)
    const dir = await api.dialog.pickFolder('选择仓库保存位置')
    if (!dir) return
    setChosenDir(dir)
    setRepoName('我的仓库')
    setStep('naming')
  }

  async function handleConfirmCreate() {
    if (!repoName.trim()) {
      setError('请输入仓库名称')
      return
    }
    setLoading(true)
    setError(null)
    try {
      const repoPath = path.join(chosenDir, `${repoName.trim()}.wg`)
      const { repoRoot, repoName: n } = await api.repo.create(repoPath, repoName.trim())
      setRepo(repoRoot, n)
    } catch (e) {
      setError((e as Error).message)
    } finally {
      setLoading(false)
    }
  }

  async function handleOpen() {
    setError(null)
    const dir = await api.dialog.pickFolder('选择仓库文件夹（*.wg）')
    if (!dir) return
    setLoading(true)
    try {
      const { repoRoot, repoName: n } = await api.repo.open(dir)
      setRepo(repoRoot, n)
    } catch (e) {
      setError((e as Error).message)
    } finally {
      setLoading(false)
    }
  }

  return (
    <div className="flex h-screen items-center justify-center bg-background">
      <Card className="w-[440px]">
        <CardHeader>
          <CardTitle className="text-2xl font-bold text-center">word_git</CardTitle>
          <p className="text-center text-muted-foreground text-sm mt-1">
            像 Git 一样管理你的 Office 文档版本
          </p>
        </CardHeader>
        <CardContent className="flex flex-col gap-4">
          {step === 'landing' && (
            <>
              <Button onClick={handleCreate} disabled={loading} size="lg" className="w-full">
                创建新仓库
              </Button>
              <Button
                onClick={handleOpen}
                disabled={loading}
                variant="outline"
                size="lg"
                className="w-full"
              >
                打开已有仓库
              </Button>
            </>
          )}

          {step === 'naming' && (
            <>
              <div className="text-sm text-muted-foreground">位置：{chosenDir}</div>
              <div className="flex flex-col gap-2">
                <Label htmlFor="repo-name">仓库名称</Label>
                <Input
                  id="repo-name"
                  placeholder="论文仓库"
                  value={repoName}
                  onChange={(e) => setRepoName(e.target.value)}
                  onKeyDown={(e) => { if (e.key === 'Enter') handleConfirmCreate() }}
                  autoFocus
                />
              </div>
              <Button onClick={handleConfirmCreate} disabled={loading} className="w-full">
                {loading ? '创建中…' : '确认创建'}
              </Button>
              <Button
                variant="ghost"
                onClick={() => setStep('landing')}
                disabled={loading}
                className="w-full"
              >
                返回
              </Button>
            </>
          )}

          {error && (
            <p className="text-destructive text-sm text-center">{error}</p>
          )}
        </CardContent>
      </Card>
    </div>
  )
}

// apps/desktop/src/renderer/src/hooks/useWatcherEvents.ts
import { useEffect } from 'react'
import { api } from '@renderer/lib/ipc'
import { useUiStore } from '@renderer/store/uiStore'

/**
 * 注册 watcher.change IPC 推送事件。
 * 放在根组件（App 或 MainLayout）中调用一次。
 */
export function useWatcherEvents(): void {
  const setPendingCommit = useUiStore((s) => s.setPendingCommit)

  useEffect(() => {
    const unsub = api.onWatcherChange(({ docName, workspacePath }) => {
      setPendingCommit(docName, workspacePath)
    })
    return unsub
  }, [setPendingCommit])
}

import { useState } from 'react'
import { useMutation, useQueryClient } from '@tanstack/react-query'
import { api } from '@renderer/lib/ipc'
import { keys } from '@renderer/lib/queryKeys'
import { useUiStore } from '@renderer/store/uiStore'

  const [message, setMessage] = useState('')
  const queryClient = useQueryClient()

      }
      setMessage('')
      clearPendingCommit()
    },
  })

  if (!pendingCommitDocName) return null

  return (

      <div className="flex items-start justify-between mb-3">
        <div>
          <p className="font-semibold text-sm">检测到文件已保存</p>
          <p className="text-xs text-muted-foreground truncate max-w-[220px]">
            {pendingCommitDocName}
          </p>
        </div>
        <Button
          size="sm"
          variant="ghost"
          className="h-7 text-xs"
          onClick={clearPendingCommit}
        >
          忽略
        </Button>
      </div>

      <Input
        placeholder="描述本次修改…"
        value={message}
        onChange={(e) => setMessage(e.target.value)}
        onKeyDown={(e) => { if (e.key === 'Enter' && !e.shiftKey) commitMutation.mutate() }}
        className="mb-2"

            {tag}
          </Badge>
        ))}
      </div>

      <Button
        className="w-full"
        onClick={() => commitMutation.mutate()}
        disabled={commitMutation.isPending}
      >
        {commitMutation.isPending ? '提交中…' : '提交新版本'}
      </Button>

      {commitMutation.isError && (
        <p className="text-destructive text-xs mt-2 text-center">
          {(commitMutation.error as Error).message}

function CommitEntry({ commit, isLatest, onRestore }: CommitEntryProps) {

export function TimelinePanel() {

          )}

          {!isLoading && (!commits || commits.length === 0) && (
            <p className="text-center text-muted-foreground text-sm py-8">暂无版本历史</p>
          )}

          {commits?.map((commit, idx) => (
            <div key={commit.id}>
              <CommitEntry
                commit={commit}
                docName={selectedDocName}
                isLatest={idx === 0}
                onRestore={() => restoreMutation.mutate(commit.id)}
              />
              {idx < commits.length - 1 && (
                <div className="ml-[22px] border-l border-dashed border-muted-foreground/30 h-2" />
              )}
            </div>
          ))}
        </div>
      </ScrollArea>
    </div>
  )
}

// apps/desktop/src/renderer/src/components/XlsxPreview.tsx
import { useEffect, useState } from 'react'
import ExcelJS from 'exceljs'

interface XlsxPreviewProps {
  data: ArrayBuffer
}

interface CellValue {
  text: string
  isHeader: boolean
}

interface SheetData {
  rows: CellValue[][]
}

export function XlsxPreview({ data }: XlsxPreviewProps) {
  const [sheetData, setSheetData] = useState<SheetData | null>(null)
  const [error, setError] = useState<string | null>(null)

  useEffect(() => {
    if (data.byteLength === 0) return
    const workbook = new ExcelJS.Workbook()
    workbook.xlsx
      .load(data)
      .then(() => {
        const sheet = workbook.worksheets[0]
        if (!sheet) {
          setError('Excel 文件中没有工作表')
          return
        }
        const rows: CellValue[][] = []
        sheet.eachRow({ includeEmpty: false }, (row, rowNum) => {
          const cells: CellValue[] = []
          row.eachCell({ includeEmpty: true }, (cell) => {
            const raw = cell.value
            let text = ''
            if (raw === null || raw === undefined) {
              text = ''
            } else if (typeof raw === 'object' && 'richText' in raw) {
              text = (raw as { richText: Array<{ text: string }> }).richText
                .map((r) => r.text)
                .join('')
            } else if (typeof raw === 'object' && 'result' in raw) {
              text = String((raw as { result: unknown }).result ?? '')
            } else {
              text = String(raw)
            }
            cells.push({ text, isHeader: rowNum === 1 })
          })
          rows.push(cells)
        })
        setSheetData({ rows })
      })
      .catch((e) => setError((e as Error).message))
  }, [data])

  if (error) {
    return <p className="text-destructive text-sm p-4">{error}</p>
  }

  if (!sheetData) {
    return <p className="text-muted-foreground text-sm p-4">解析中…</p>
  }

  return (
    <div className="overflow-auto h-full p-4">
      <table className="text-sm border-collapse w-full">
        <tbody>
          {sheetData.rows.map((row, ri) => (
            <tr key={ri}>
              {row.map((cell, ci) =>
                cell.isHeader ? (
                  <th
                    key={ci}
                    className="border border-border px-2 py-1 bg-muted font-semibold text-left"
                  >
                    {cell.text}
                  </th>
                ) : (
                  <td key={ci} className="border border-border px-2 py-1">
                    {cell.text}
                  </td>
                ),
              )}
            </tr>
          ))}
        </tbody>
      </table>
    </div>
  )
}

import { keys } from '@renderer/lib/queryKeys'
import { useUiStore } from '@renderer/store/uiStore'

import { Button } from '@renderer/components/ui/button'

  const [scrubResult, setScrubResult] = useState<ScrubResult | null>(null)

  const { data: stats } = useQuery({
    queryKey: keys.storageStats(),
    queryFn: () => api.stats.storage(),
  })

  const { data: quarantineEntries } = useQuery({
    queryKey: keys.quarantineList(),
    queryFn: () => api.maintenance.quarantine.list(),
  })

  const scrubMutation = useMutation({
    mutationFn: () => api.maintenance.scrub(),
    onSuccess: (result) => setScrubResult(result),

    mutationFn: async () => {
      const destPath = await api.dialog.savePath('选择备份保存位置', 'wordgit-backup.wgbak')

      return api.backup.export(destPath)
    },

  const importMutation = useMutation({
    mutationFn: async () => {
      const srcPath = await api.dialog.pickFile('选择备份文件', [
        { name: 'word_git 备份', extensions: ['wgbak', 'zip'] },
      ])
      if (!srcPath) return
      const destDir = await api.dialog.pickFolder('选择恢复到的位置')
      if (!destDir) return
      await api.backup.import(srcPath, destDir)

      <div className="flex items-center justify-between px-4 py-3 border-b border-border shrink-0">

            <h3 className="text-sm font-semibold mb-3">存储统计</h3>

          </section>

              variant="outline"
              size="sm"

              备份导出

              variant="outline"
              size="sm"

              从备份恢复

  )
}
